def create_empty_circuit(n_qubits, n_clbits):
    qr = QuantumRegister(n_qubits, "q")
    if n_clbits > 0:
        return QuantumCircuit(qr, ClassicalRegister(n_clbits, "c"))
    return QuantumCircuit(qr)

def apply_gate(qc, gate_name, qubits, params=None):
    g = gate_name.upper()
    if g == "H": qc.h(qubits[0])
    elif g == "X": qc.x(qubits[0])
    elif g == "Y": qc.y(qubits[0])
    elif g == "Z": qc.z(qubits[0])
    elif g == "S": qc.s(qubits[0])
    elif g == "SDG": qc.sdg(qubits[0])
    elif g == "T": qc.t(qubits[0])
    elif g == "TDG": qc.tdg(qubits[0])
    elif g == "RX": qc.rx(params[0], qubits[0])
    elif g == "RY": qc.ry(params[0], qubits[0])
    elif g == "RZ": qc.rz(params[0], qubits[0])
    elif g in ("CNOT", "CX"): qc.cx(qubits[0], qubits[1])
    elif g == "CZ": qc.cz(qubits[0], qubits[1])
    elif g == "SWAP": qc.swap(qubits[0], qubits[1])
    elif g in ("CCX", "TOFFOLI"): qc.ccx(qubits[0], qubits[1], qubits[2])
    elif g == "MEASURE":
        if qc.num_clbits > qubits[0]:
            qc.measure(qubits[0], qubits[0])
    elif g == "BARRIER": qc.barrier()
    elif g == "RESET": qc.reset(qubits[0])
    elif g == "QFT":
        qc.append(QFT(len(qubits), do_swaps=True).to_instruction(), qubits)

def rebuild_circuit_from_history():
    qc = create_empty_circuit(st.session_state.n_qubits, st.session_state.n_clbits)
    for e in st.session_state.gate_history:
        apply_gate(qc, e["gate"], e["qubits"], e.get("params"))
    st.session_state.circuit = qc
    return qc

def add_gate(gate_name, qubits, params=None):
    st.session_state.gate_history.append({
        "gate": gate_name, "qubits": list(qubits),
        "params": list(params) if params else None
    })
    rebuild_circuit_from_history()

def reset_all():
    st.session_state.gate_history = []
    st.session_state.last_counts = None
    st.session_state.last_statevector = None
    rebuild_circuit_from_history()

def build_readout_calibration_matrix(n_qubits, p1given0=0.03, p0given1=0.05):
    """
    Matriz de calibración de lectura (confusion matrix) por producto tensorial.

    Convención A[i, j] = P(medir i | preparado j)
    - p1given0: P(1|0)  error al medir 1 cuando era 0
    - p0given1: P(0|1)  error al medir 0 cuando era 1

    Para 1 qubit:
        [[P(0|0), P(0|1)],
         [P(1|0), P(1|1)]]
      = [[1-p1given0, p0given1],
         [p1given0,   1-p0given1]]
    """
    # Matriz 1-qubit (filas = medido, columnas = preparado)
    a00 = 1.0 - p1given0
    a01 = p0given1
    a10 = p1given0
    a11 = 1.0 - p0given1
    single = np.array([[a00, a01], [a10, a11]], dtype=float)
    # Producto tensorial para n qubits (qubits independientes)
    conf = np.array([[1.0]])
    for _ in range(n_qubits):
        conf = np.kron(conf, single)
    return conf


def apply_readout_mitigation(counts, n_qubits, error_rate=0.04,
                             p1given0=None, p0given1=None, method="tensored"):
    """
    Mitigación de errores de lectura (measurement/readout error mitigation).

    method:
      - "tensored": matriz por producto tensorial de errores 1-qubit (recomendado)
      - "uniform":  modelo antiguo (error uniforme hacia todos los demás estados)

    Si p1given0 / p0given1 no se pasan, se derivan de error_rate
    (asimetría típica: P(0|1) un poco mayor que P(1|0)).
    """
    if p1given0 is None:
        p1given0 = float(error_rate)
    if p0given1 is None:
        p0given1 = float(min(error_rate * 1.4, 0.49))  # asimetría suave

    dim = 2 ** n_qubits
    labels = [format(i, f"0{n_qubits}b") for i in range(dim)]
    total = sum(counts.values()) or 1
    p_meas = np.array([counts.get(lab, 0) / total for lab in labels], dtype=float)

    if method == "uniform":
        conf = np.eye(dim) * (1 - error_rate) + (error_rate / max(dim - 1, 1)) * (1 - np.eye(dim))
    else:
        conf = build_readout_calibration_matrix(n_qubits, p1given0=p1given0, p0given1=p0given1)

    try:
        inv = np.linalg.pinv(conf)
    except Exception:
        return dict(counts)

    p_corr = inv @ p_meas
    # Proyección a probabilidad (clip + renormalizar) — estabiliza inversas mal condicionadas
    p_corr = np.clip(p_corr, 0.0, None)
    s = p_corr.sum()
    if s > 0:
        p_corr = p_corr / s
    else:
        p_corr = p_meas

    mitigated = {lab: int(round(float(p_corr[i]) * total)) for i, lab in enumerate(labels)}
    # Ajuste fino para conservar el total de shots
    diff = total - sum(mitigated.values())
    if diff != 0 and labels:
        # Asignar la diferencia al estado de mayor probabilidad corregida
        k = int(np.argmax(p_corr))
        mitigated[labels[k]] = mitigated.get(labels[k], 0) + int(diff)
    return mitigated


def readout_mitigation_report(counts_raw, counts_mit, n_qubits):
    """Resumen comparativo raw vs mitigado para la UI."""
    total_r = sum(counts_raw.values()) or 1
    total_m = sum(counts_mit.values()) or 1
    keys = sorted(set(counts_raw) | set(counts_mit), key=lambda s: int(s, 2) if set(s) <= set("01") else s)
    rows = []
    for k in keys:
        pr = counts_raw.get(k, 0) / total_r
        pm = counts_mit.get(k, 0) / total_m
        rows.append({"estado": k, "p_raw": pr, "p_mit": pm, "delta": pm - pr})
    return rows


def run_automatic_readout_calibration(n_qubits, shots=2048, p1given0_true=0.03, p0given1_true=0.05,
                                      method="tensored"):
    """
    Calibración automática de lectura.

    Prepara estados base conocidos, los mide con un modelo de error de lectura
    (ReadoutError de Aer) y estima la matriz de confusión.

    method:
      - "tensored": calibra solo |0> y |1> (1 qubit) y construye el producto tensorial
                    (barato, escalable)
      - "full":     calibra los 2^n estados computacionales (costoso; solo n pequeño)

    Devuelve: (conf_matrix numpy, info dict)
    """
    from qiskit_aer import AerSimulator
    from qiskit_aer.noise import NoiseModel, ReadoutError

    # Error de lectura "físico" inyectado en la calibración (simula dispositivo)
    re = ReadoutError([[1 - p1given0_true, p1given0_true],
                       [p0given1_true, 1 - p0given1_true]])
    nm = NoiseModel()
    for q in range(n_qubits):
        nm.add_readout_error(re, [q])
    sim = AerSimulator(noise_model=nm)

    def measure_prep(bitstring):
        qc = QuantumCircuit(n_qubits, n_qubits)
        for i, b in enumerate(bitstring):
            if b == "1":
                qc.x(i)
        qc.measure(range(n_qubits), range(n_qubits))
        counts = sim.run(transpile(qc, sim), shots=shots).result().get_counts()
        return counts

    labels = [format(i, f"0{n_qubits}b") for i in range(2 ** n_qubits)]

    if method == "full" and n_qubits <= 4:
        # A[i,j] = P(medir i | preparado j)
        conf = np.zeros((2 ** n_qubits, 2 ** n_qubits), dtype=float)
        for j, prep in enumerate(labels):
            counts = measure_prep(prep)
            total = sum(counts.values()) or 1
            for i, meas in enumerate(labels):
                conf[i, j] = counts.get(meas, 0) / total
        info = {"method": "full", "n_qubits": n_qubits, "shots_per_state": shots,
                "states_calibrated": len(labels)}
    else:
        # Tensored: calibrar qubit 0 en un registro de n qubits preparando 0...0 y 1000...
        # Equivalente a estimar P(0|0), P(1|0), P(0|1), P(1|1) en un qubit
        # Usamos el qubit 0 para la estadística agregada
        counts0 = measure_prep("0" * n_qubits)
        counts1 = measure_prep("1" + "0" * (n_qubits - 1))
        total0 = sum(counts0.values()) or 1
        total1 = sum(counts1.values()) or 1
        # P(medir 0 en q0 | prep 0) = suma counts con bit0=0
        def bit0_prob(counts, bit_val, total):
            s = 0
            for k, v in counts.items():
                # Qiskit bitstring: qubit 0 es el carácter de la derecha
                if len(k) >= 1 and k[-1] == bit_val:
                    s += v
            return s / total
        # Para prep |0>: qubit 0 preparado en 0
        p0_given0 = bit0_prob(counts0, "0", total0)
        p1_given0 = bit0_prob(counts0, "1", total0)
        # Para prep |1> en qubit 0:
        p0_given1 = bit0_prob(counts1, "0", total1)
        p1_given1 = bit0_prob(counts1, "1", total1)
        # Normalizar por seguridad
        s0 = p0_given0 + p1_given0
        s1 = p0_given1 + p1_given1
        if s0 > 0:
            p0_given0, p1_given0 = p0_given0 / s0, p1_given0 / s0
        if s1 > 0:
            p0_given1, p1_given1 = p0_given1 / s1, p1_given1 / s1
        conf = build_readout_calibration_matrix(
            n_qubits, p1given0=p1_given0, p0given1=p0_given1
        )
        info = {
            "method": "tensored",
            "n_qubits": n_qubits,
            "shots_per_state": shots,
            "P(0|0)": float(p0_given0),
            "P(1|0)": float(p1_given0),
            "P(0|1)": float(p0_given1),
            "P(1|1)": float(p1_given1),
        }

    return conf, info


def apply_readout_mitigation_with_matrix(counts, n_qubits, conf_matrix):
    """Aplica mitigación usando una matriz de calibración ya estimada."""
    dim = 2 ** n_qubits
    labels = [format(i, f"0{n_qubits}b") for i in range(dim)]
    total = sum(counts.values()) or 1
    p_meas = np.array([counts.get(lab, 0) / total for lab in labels], dtype=float)
    conf = np.asarray(conf_matrix, dtype=float)
    if conf.shape != (dim, dim):
        # fallback
        return apply_readout_mitigation(counts, n_qubits)
    try:
        inv = np.linalg.pinv(conf)
    except Exception:
        return dict(counts)
    p_corr = inv @ p_meas
    p_corr = np.clip(p_corr, 0.0, None)
    s = p_corr.sum()
    if s > 0:
        p_corr = p_corr / s
    mitigated = {lab: int(round(float(p_corr[i]) * total)) for i, lab in enumerate(labels)}
    diff = total - sum(mitigated.values())
    if diff != 0 and labels:
        k = int(np.argmax(p_corr))
        mitigated[labels[k]] = mitigated.get(labels[k], 0) + int(diff)
    return mitigated


def simple_zne_extrapolation(values, factors, method="linear"):
    """
    Extrapolación a ruido cero.
    method: "linear" | "richardson" (polinomio grado 2 cuando hay >=3 puntos)
    """
    values = np.asarray(values, dtype=float)
    factors = np.asarray(factors, dtype=float)
    if len(values) < 2:
        return float(values[0]) if len(values) else 0.0
    if method == "richardson" and len(values) >= 3:
        # Ajuste cuadrático (Richardson de orden 2)
        coeffs = np.polyfit(factors, values, 2)
        return float(coeffs[-1])  # término independiente (lambda=0)
    # Lineal por defecto
    coeffs = np.polyfit(factors, values, 1)
    return float(coeffs[-1])

def run_local_with_mitigation(qc, shots=1024, noise=True, readout_mit=True, zne=False,
                               zne_method="linear", p1given0=0.03, p0given1=0.05,
                               conf_matrix=None):
    """
    Ejecución local con mitigación opcional.
    Si conf_matrix está definida, se usa calibración automática.
    Devuelve: counts_mitigados, zne_value, all_results, exp_values, counts_raw
    """
    from qiskit_aer.noise import NoiseModel, depolarizing_error
    sim = AerSimulator()
    factors = [1.0, 3.0, 5.0] if zne else [1.0]
    all_results = {}
    exp_values = []
    counts_raw_base = {}
    for factor in factors:
        noise_model = None
        if noise:
            p1 = min(0.012 * factor, 0.45)
            p2 = min(0.025 * factor, 0.45)
            nm = NoiseModel()
            nm.add_all_qubit_quantum_error(depolarizing_error(p1, 1),
                ["h","x","y","z","s","t","rx","ry","rz","sdg","tdg"])
            nm.add_all_qubit_quantum_error(depolarizing_error(p2, 2), ["cx","cz","swap"])
            noise_model = nm
        job = sim.run(transpile(qc, sim), shots=shots, noise_model=noise_model)
        counts_raw = job.result().get_counts()
        if factor == 1.0:
            counts_raw_base = dict(counts_raw)
        if readout_mit:
            if conf_matrix is not None:
                counts = apply_readout_mitigation_with_matrix(
                    counts_raw, qc.num_qubits, conf_matrix
                )
            else:
                counts = apply_readout_mitigation(
                    counts_raw, qc.num_qubits,
                    p1given0=p1given0, p0given1=p0given1, method="tensored"
                )
        else:
            counts = counts_raw
        all_results[factor] = counts
        total = sum(counts.values()) or 1
        exp = sum((1 if bitstr[-1] == "0" else -1) * cnt for bitstr, cnt in counts.items()) / total
        exp_values.append(exp)
    if zne:
        zne_val = simple_zne_extrapolation(exp_values, factors, method=zne_method)
    else:
        zne_val = exp_values[0] if exp_values else 0.0
    return all_results.get(1.0, {}), float(zne_val), all_results, exp_values, counts_raw_base

def spsa_minimize(fun, x0, maxiter=40, a=0.2, c=0.1):
    theta = np.array(x0, dtype=float)
    history = []
    for k in range(1, maxiter + 1):
        ak = a / (k ** 0.602)
        ck = c / (k ** 0.101)
        delta = np.random.choice([-1., 1.], size=theta.shape)
        f_plus = fun(theta + ck * delta)
        f_minus = fun(theta - ck * delta)
        ghat = (f_plus - f_minus) / (2.0 * ck * delta)
        theta = theta - ak * ghat
        history.append(float(fun(theta)))
    return theta, history

def get_hamiltonian(n_qubits, ham_type="ising"):
    """
    Hamiltonianos de demostración orientados a investigación.
    Son modelos simplificados (no simulaciones de alta fidelidad).
    """
    # Normalizar nombres de investigación a familias base
    research_map = {
        "ising": "ising",
        "heisenberg": "heisenberg",
        "molecular": "molecular",
        # Temas de investigación (mapeados a modelos base honestos)
        "metamaterial_spin": "heisenberg",      # espines / YIG-like
        "propulsion_opt": "ising",              # optimización de recursos / thruster
        "he3_interaction": "heisenberg",        # interacción tipo espín / materia
        "nanomedicine_mol": "molecular",        # estructura electrónica simplificada
    }
    base = research_map.get(ham_type, "ising")

    if base == "ising":
        if n_qubits == 2:
            return SparsePauliOp.from_list([("ZZ", 1.0), ("XI", 0.5), ("IX", 0.5)])
        return SparsePauliOp.from_list([
            ("ZZ" + "I"*(n_qubits-2), 1.0),
            ("X" + "I"*(n_qubits-1), 0.4)
        ])
    elif base == "heisenberg":
        ops = []
        for i in range(n_qubits - 1):
            xx = ["I"] * n_qubits
            yy = ["I"] * n_qubits
            zz = ["I"] * n_qubits
            xx[i] = "X"; xx[i+1] = "X"
            yy[i] = "Y"; yy[i+1] = "Y"
            zz[i] = "Z"; zz[i+1] = "Z"
            ops.append(("".join(xx), 0.5))
            ops.append(("".join(yy), 0.5))
            ops.append(("".join(zz), 0.8))
        return SparsePauliOp.from_list(ops)
    else:  # molecular
        if n_qubits == 2:
            return SparsePauliOp.from_list([("II", -1.0), ("ZZ", 0.7), ("XX", 0.3), ("YY", 0.3)])
        return SparsePauliOp.from_list([
            ("Z" + "I"*(n_qubits-1), -0.8),
            ("ZZ" + "I"*(n_qubits-2), 0.5),
            ("X" + "I"*(n_qubits-1), 0.2)
        ])

def run_vqe_with_optimizer(n_qubits, optimizer_name, maxiter=40, learning_rate=0.05, ham_type="ising"):
    hamiltonian = get_hamiltonian(n_qubits, ham_type)
    ansatz = EfficientSU2(n_qubits, reps=1, entanglement="linear")
    estimator = StatevectorEstimator()

    def energy(params):
        bound = ansatz.assign_parameters(params)
        return float(estimator.run([(bound, hamiltonian)]).result()[0].data.evs)

    x0 = np.random.uniform(-np.pi, np.pi, ansatz.num_parameters)
    history = []

    if optimizer_name == "SPSA":
        theta_opt, history = spsa_minimize(energy, x0, maxiter=maxiter)
        return energy(theta_opt), theta_opt, history, ansatz
    elif optimizer_name == "ADAM":
        theta = x0.copy()
        m = np.zeros_like(theta)
        v = np.zeros_like(theta)
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        for t in range(1, maxiter + 1):
            grad = np.zeros_like(theta)
            f0 = energy(theta)
            eps_fd = 1e-4
            for i in range(len(theta)):
                th = theta.copy()
                th[i] += eps_fd
                grad[i] = (energy(th) - f0) / eps_fd
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            theta = theta - learning_rate * m_hat / (np.sqrt(v_hat) + eps)
            history.append(float(energy(theta)))
        return energy(theta), theta, history, ansatz
    else:
        method_map = {
            "COBYLA": "COBYLA", "L-BFGS-B": "L-BFGS-B", "SLSQP": "SLSQP",
            "Nelder-Mead": "Nelder-Mead", "Powell": "Powell"
        }
        method = method_map.get(optimizer_name, "COBYLA")
        def cb(xk):
            history.append(float(energy(xk)))
        res = minimize(energy, x0, method=method, options={"maxiter": maxiter},
                       callback=cb if method != "COBYLA" else None)
        if not history:
            history = [float(res.fun)]
        return res.fun, res.x, history, ansatz

def _build_depolarizing_noise(factor=1.0):
    """Modelo de ruido depolarizing escalado por factor (para ZNE)."""
    from qiskit_aer.noise import NoiseModel, depolarizing_error
    p1 = min(0.012 * factor, 0.45)
    p2 = min(0.025 * factor, 0.45)
    nm = NoiseModel()
    single = ["h", "x", "y", "z", "s", "t", "rx", "ry", "rz", "sdg", "tdg", "id"]
    double = ["cx", "cz", "swap", "rzz", "cp"]
    nm.add_all_qubit_quantum_error(depolarizing_error(p1, 1), single)
    nm.add_all_qubit_quantum_error(depolarizing_error(p2, 2), double)
    return nm


def energy_with_real_zne(params, ansatz, hamiltonian, factors=(1.0, 3.0, 5.0),
                         zne_method="linear"):
    """
    Evaluación de energía con ZNE real:
      1) Simula el estado con ruido depolarizing a varios factores λ
      2) Calcula ⟨H⟩ en cada λ (density matrix)
      3) Extrapola a λ → 0
    Devuelve: (energia_mitigada, lista_valores_por_factor)
    """
    from qiskit_aer import AerSimulator
    from qiskit.quantum_info import DensityMatrix

    bound = ansatz.assign_parameters(params)
    # Circuito sin medidas para density matrix
    qc = bound.copy()
    try:
        qc.save_density_matrix()
    except Exception:
        pass

    obs_matrix = hamiltonian.to_matrix()
    values = []
    for f in factors:
        nm = _build_depolarizing_noise(f)
        sim = AerSimulator(method="density_matrix", noise_model=nm)
        tqc = transpile(qc, sim)
        result = sim.run(tqc).result()
        # Extraer densidad
        try:
            rho = result.data(0)["density_matrix"]
        except Exception:
            rho = list(result.data(0).values())[0]
        # ⟨H⟩ = Re(Tr(ρ H))
        exp_val = float(np.real(np.trace(rho @ obs_matrix)))
        values.append(exp_val)

    mitigated = simple_zne_extrapolation(values, list(factors), method=zne_method)
    return float(mitigated), values


def cam_vqe(n_qubits=2, total_budget=70, ham_type="ising",
            mitigation_mode="standard", use_real_zne=False, zne_method="linear"):
    """
    CAM-VQE v1.8 — Canelone Adaptive Mitigation VQE

    mitigation_mode:
      - "standard": proxy de ruido normal
      - "zne_aware": umbrales más sensibles (filosofía ZNE en el meta-controlador)

    use_real_zne:
      - False: energía exacta (StatevectorEstimator) — rápido, prototipo
      - True:  ZNE real en cada evaluación de energía
               (ruido depolarizing a λ=1,3,5 + extrapolación a λ=0)
               Más costoso computacionalmente; presupuesto efectivo menor.
    """
    hamiltonian = get_hamiltonian(n_qubits, ham_type)
    ansatz = EfficientSU2(n_qubits, reps=1, entanglement="linear")
    estimator = StatevectorEstimator()
    n_params = ansatz.num_parameters
    theta = np.random.uniform(-np.pi, np.pi, n_params)

    history = []
    phase_log = []
    current_phase = "exploration"
    best_energy = float("inf")
    best_theta = theta.copy()
    noise_history = []
    zne_raw_history = []  # valores pre-extrapolación cuando use_real_zne

    def energy_ideal(params):
        bound = ansatz.assign_parameters(params)
        return float(estimator.run([(bound, hamiltonian)]).result()[0].data.evs)

    def energy(params):
        if not use_real_zne:
            return energy_ideal(params)
        mitigated, raw_vals = energy_with_real_zne(
            params, ansatz, hamiltonian,
            factors=(1.0, 3.0, 5.0),
            zne_method=zne_method
        )
        zne_raw_history.append({"params_norm": float(np.linalg.norm(params)), "raw": raw_vals, "mitigated": mitigated})
        return mitigated

    def estimate_noise_proxy(params):
        """Proxy de ruido: si ZNE real está activo, usa dispersión de los factores;
        si no, usa perturbaciones del paisaje ideal."""
        if use_real_zne:
            # Usar la última evaluación ZNE o una nueva
            _, raw = energy_with_real_zne(params, ansatz, hamiltonian,
                                          factors=(1.0, 3.0, 5.0), zne_method=zne_method)
            slope = abs(raw[-1] - raw[0]) / 4.0
            variance = float(np.var(raw))
            proxy = 0.6 * min(slope * 2, 1.0) + 0.4 * min(variance * 5, 1.0)
            return float(np.clip(proxy, 0.0, 1.0))
        vals = []
        for scale in [1.0, 1.8, 2.6]:
            pert = params + np.random.normal(0, 0.04 * scale, size=params.shape)
            vals.append(energy_ideal(pert))
        slope = abs(vals[-1] - vals[0]) / 1.6
        variance = np.var(vals)
        proxy = 0.6 * min(slope, 1.0) + 0.4 * min(variance * 5, 1.0)
        return float(np.clip(proxy, 0.0, 1.0))

    step = 0
    stagnation = 0
    prev_best = best_energy

    while step < total_budget:
        noise_est = estimate_noise_proxy(theta)
        noise_history.append(noise_est)

        # Meta-controlador (umbrales según mitigation_mode)
        if mitigation_mode == "zne_aware":
            # Más sensible al ruido: se mantiene más tiempo en exploración (espíritu ZNE)
            th_high, th_mid = 0.18, 0.08
            explore_bias = 0.45
        else:
            th_high, th_mid = 0.28, 0.12
            explore_bias = 0.35

        if noise_est > th_high or (current_phase == "exploration" and step < total_budget * explore_bias):
            current_phase = "exploration"
            a = 0.22
            c = 0.11
            delta = np.random.choice([-1., 1.], size=theta.shape)
            f_plus = energy(theta + c * delta)
            f_minus = energy(theta - c * delta)
            ghat = (f_plus - f_minus) / (2.0 * c * delta)
            theta = theta - a * ghat
            e = energy(theta)
            step += 2

        elif noise_est > th_mid:
            current_phase = "refinement"
            grad = np.zeros_like(theta)
            f0 = energy(theta)
            eps_fd = 1e-4
            for i in range(len(theta)):
                th = theta.copy()
                th[i] += eps_fd
                grad[i] = (energy(th) - f0) / eps_fd
            # Learning rate adaptativo al ruido
            lr = 0.09 * (1.0 - 0.7 * noise_est)
            theta = theta - lr * grad
            e = energy(theta)
            step += len(theta) + 1

        else:
            current_phase = "polishing"
            def local_fun(x):
                return energy(x)
            res = minimize(local_fun, theta, method="L-BFGS-B",
                           options={"maxiter": 6, "ftol": 1e-9})
            theta = res.x
            e = float(res.fun)
            step += 6

        history.append({
            "step": step,
            "energy": float(e),
            "phase": current_phase,
            "noise_est": float(noise_est)
        })
        phase_log.append(current_phase)

        if e < best_energy - 1e-8:
            best_energy = e
            best_theta = theta.copy()
            stagnation = 0
        else:
            stagnation += 1

        # Criterio de parada mejorado
        if step > 20 and stagnation >= 8:
            break
        if step > 25 and len(history) >= 6:
            recent = [h["energy"] for h in history[-6:]]
            if max(recent) - min(recent) < 5e-5:
                break

    return {
        "best_energy": best_energy,
        "best_params": best_theta.tolist(),
        "history": history,
        "phase_log": phase_log,
        "noise_history": noise_history,
        "ansatz": ansatz,
        "hamiltonian_type": ham_type,
        "n_qubits": n_qubits,
        "mitigation_mode": mitigation_mode,
        "use_real_zne": use_real_zne,
        "zne_method": zne_method if use_real_zne else None,
        "zne_raw_samples": len(zne_raw_history),
    }
